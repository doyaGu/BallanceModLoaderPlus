cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BMLCryptoTests LANGUAGES C CXX)
    get_filename_component(PROJECT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    set(BML_SOURCE_DIR "${PROJECT_SOURCE_DIR}/src")
    set(BML_RUNTIME_SDL 1)
    set(CMAKE_CXX_STANDARD 20)
    find_package(SDL3 CONFIG REQUIRED)
    add_subdirectory("${PROJECT_SOURCE_DIR}/deps" deps)
    add_subdirectory("${BML_SOURCE_DIR}/Utils" utils)
    enable_testing()
    if(BML_GTEST_SOURCE)
        set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
        add_subdirectory("${BML_GTEST_SOURCE}" googletest)
    else()
        find_package(GTest CONFIG REQUIRED)
    endif()
endif()

set(_signing_sources SigningTest.cpp "${PROJECT_SOURCE_DIR}/tools/updater/UpdaterSigning.cpp")
add_executable(BMLNativeCryptoTest CryptoTest.cpp ${_signing_sources})
target_compile_features(BMLNativeCryptoTest PRIVATE cxx_std_20)
target_include_directories(BMLNativeCryptoTest PRIVATE "${BML_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/tools/updater")
target_link_libraries(BMLNativeCryptoTest PRIVATE BML_CryptoUtils BML_PathUtils BML_P256 GTest::gtest_main)
add_test(NAME BMLNativeCryptoTest COMMAND BMLNativeCryptoTest)

# Exercise the Linux SHA/P-256 implementation on every host as well.
add_library(BMLPortableCrypto STATIC "${BML_SOURCE_DIR}/Utils/CryptoUtilsPosix.cpp"
    "${BML_SOURCE_DIR}/Utils/CryptoVerifyPortable.cpp")
target_compile_features(BMLPortableCrypto PUBLIC cxx_std_20)
target_compile_definitions(BMLPortableCrypto PRIVATE BML_CRYPTO_PORTABLE=1)
target_include_directories(BMLPortableCrypto PUBLIC "${BML_SOURCE_DIR}/Utils")
target_link_libraries(BMLPortableCrypto PUBLIC BML_Sha256 BML_P256 BML_StringUtils SDL3::SDL3)
add_executable(BMLPortableCryptoTest CryptoTest.cpp ${_signing_sources})
target_include_directories(BMLPortableCryptoTest PRIVATE "${BML_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/tools/updater")
target_link_libraries(BMLPortableCryptoTest PRIVATE BMLPortableCrypto BML_PathUtils GTest::gtest_main)
add_test(NAME BMLPortableCryptoTest COMMAND BMLPortableCryptoTest)
set_tests_properties(BMLNativeCryptoTest BMLPortableCryptoTest PROPERTIES LABELS "crypto;platform")
add_executable(BMLSignerRandomFailureTest RandomFailureTest.cpp "${PROJECT_SOURCE_DIR}/tools/updater/UpdaterSigning.cpp")
target_compile_features(BMLSignerRandomFailureTest PRIVATE cxx_std_20)
target_include_directories(BMLSignerRandomFailureTest PRIVATE "${PROJECT_SOURCE_DIR}/tools/updater")
target_link_libraries(BMLSignerRandomFailureTest PRIVATE BML_P256 GTest::gtest_main)
add_test(NAME BMLSignerRandomFailureTest COMMAND BMLSignerRandomFailureTest)
