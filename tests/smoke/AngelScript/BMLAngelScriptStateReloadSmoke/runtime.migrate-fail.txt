void TraceStateHookPhase(BML::ReloadPhase expected, const string &in label) {
  BML::ModContext current;
  if (BML::BorrowCurrentContext(current)) {
    const bool phaseOk = current.IsReloading && current.ReloadPhase == expected;
    current.LogInfo("BML state hook phase: " + label + "=" + (phaseOk ? "valid" : "unexpected"));
  }
}

[bml.mod id="bml.state.reload.smoke" name="BML AngelScript State Reload Smoke" version="2.0.0" author="BML" bml="0.3.0" reload="auto" description="Smoke test for script hot reload migration failure."]
class BMLStateReloadSmokeMod {
  void MigrateState(const string &in version, BML::StateBag@ state) {
    TraceStateHookPhase(BML::RELOAD_MIGRATE_STATE, "migrate-fail migrate");
    BML::ModContext current;
    if (!BML::BorrowCurrentContext(current)) {
      throw("migrate-fail could not borrow current context");
    }
    current.SkipRenderForNextTick();
    throw("intentional state reload migrate failure smoke");
  }

  void RestoreState(BML::StateBag@ state) {
    if (state !is null) {
      state.SetString("restoreTouched", "true");
    }
  }

  void OnLoad(const BML::ModContext &in ctx) {
    BML::Logger@ logger = ctx.BorrowLogger();
    if (logger !is null) {
      logger.Info("BML state reload smoke migrate-fail candidate should not load");
    }
  }

  void OnUnload(const BML::ModContext &in ctx) {
    BML::Logger@ logger = ctx.BorrowLogger();
    if (logger !is null) {
      const bool cleanupPhaseOk = ctx.IsReloading && ctx.ReloadPhase == BML::RELOAD_CLEANUP;
      logger.Info("BML failed candidate cleanup phase: migrate=" + (cleanupPhaseOk ? "valid" : "unexpected"));
    }
  }
}
