void TraceStateHookPhase(BML::ReloadPhase expected, const string &in label) {
  BML::ModContext current;
  if (BML::BorrowCurrentContext(current)) {
    const bool phaseOk = current.IsReloading && current.ReloadPhase == expected;
    current.LogInfo("BML state hook phase: " + label + "=" + (phaseOk ? "valid" : "unexpected"));
  }
}

[bml.mod id="bml.state.reload.smoke" name="BML AngelScript State Reload Smoke" version="2.0.0" author="BML" bml="0.3.0" reload="auto" description="Smoke test for script hot reload state migration."]
class BMLStateReloadSmokeMod {
  bool migrated = false;
  int counter = -1;
  int frames = 0;
  string text = "";
  string fromVersion = "";
  bool requestedExit = false;

  void MigrateState(const string &in version, BML::StateBag@ state) {
    TraceStateHookPhase(BML::RELOAD_MIGRATE_STATE, "v2 migrate");
    if (state is null) {
      return;
    }
    migrated = state.GetBool("saved", false);
    counter = state.GetInt("counter", -1) + 1;
    text = state.GetString("text", "") + ":migrated";
    fromVersion = version;
    state.SetInt("counter", counter);
    state.SetString("text", text);
    state.SetString("fromVersion", fromVersion);
  }

  void RestoreState(BML::StateBag@ state) {
    TraceStateHookPhase(BML::RELOAD_RESTORE_STATE, "v2 restore");
    if (state is null) {
      return;
    }
    counter = state.GetInt("counter", -1);
    text = state.GetString("text", "");
    fromVersion = state.GetString("fromVersion", "");
  }

  void OnLoad(const BML::ModContext &in ctx) {
    BML::Logger@ logger = ctx.BorrowLogger();
    if (logger !is null) {
      const bool reloadPhaseOk = ctx.IsReloading && ctx.ReloadPhase == BML::RELOAD_LOAD;
      logger.Info("BML state reload phase: v2 load=" + (reloadPhaseOk ? "reload" : "unexpected"));
      logger.Info("BML state reload smoke v2 loaded migrated=" + (migrated ? "true" : "false") +
                  " from=" + fromVersion +
                  " counter=" + counter +
                  " text=" + text);
    }
  }

  void OnProcess(const BML::ModContext &in ctx) {
    ++frames;
    ImGui::SetNextWindowPos(ImVec2(64.0f, 64.0f), ImGuiCond_Once);
    ImGui::SetNextWindowSize(ImVec2(320.0f, 0.0f), ImGuiCond_Once);
    if (ImGui::Begin("State Reload Smoke")) {
      ImGui::TextUnformatted("runtime v2");
      ImGui::TextUnformatted("migrated=" + (migrated ? "true" : "false"));
      ImGui::TextUnformatted("counter=" + counter);
    }
    ImGui::End();
    if (frames >= 30 && !requestedExit) {
      requestedExit = true;
      ctx.ExecuteCommand("exit");
    }
  }

  void OnUnload(const BML::ModContext &in ctx) {
    BML::Logger@ logger = ctx.BorrowLogger();
    if (logger !is null) {
      const bool shutdownPhaseOk = !ctx.IsReloading && ctx.ReloadPhase == BML::RELOAD_NONE;
      logger.Info("BML state reload phase: v2 unload=" + (shutdownPhaseOk ? "shutdown" : "unexpected"));
    }
  }
}
