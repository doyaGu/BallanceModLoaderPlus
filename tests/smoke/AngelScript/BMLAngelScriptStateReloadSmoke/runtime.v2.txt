void TraceStateHookPhase(BML::ReloadPhase expected, const string &in label) {
  BML::ModContext current;
  if (BML::BorrowCurrentContext(current)) {
    const bool phaseOk = current.IsReloading && current.ReloadPhase == expected;
    current.LogInfo("BML state hook phase: " + label + "=" + (phaseOk ? "valid" : "unexpected"));
  }
}

[bml.mod id="bml.state.reload.smoke" name="BML AngelScript State Reload Smoke" version="2.0.0" author="BML" bml="0.3.0" reload="auto" description="Smoke test for script hot reload state migration."]
class BMLStateReloadSmokeMod {
  bool migrated = false;
  int counter = -1;
  int frames = 0;
  string text = "";
  string fromVersion = "";
  bool requestedCommand = false;
  bool publishedDataShare = false;
  bool requestedExit = false;
  BML::TimerRef@ reloadTimer;
  BML::CommandRef@ reloadCommand;
  BML::DataShareRequestRef@ reloadDataShare;

  void MigrateState(const string &in version, BML::StateBag@ state) {
    TraceStateHookPhase(BML::RELOAD_MIGRATE_STATE, "v2 migrate");
    if (state is null) {
      return;
    }
    migrated = state.GetBool("saved", false);
    counter = state.GetInt("counter", -1) + 1;
    text = state.GetString("text", "") + ":migrated";
    fromVersion = version;
    state.SetInt("counter", counter);
    state.SetString("text", text);
    state.SetString("fromVersion", fromVersion);
  }

  void RestoreState(BML::StateBag@ state) {
    TraceStateHookPhase(BML::RELOAD_RESTORE_STATE, "v2 restore");
    if (state is null) {
      return;
    }
    counter = state.GetInt("counter", -1);
    text = state.GetString("text", "");
    fromVersion = state.GetString("fromVersion", "");
  }

  void OnLoad(const BML::ModContext &in ctx) {
    BML::Logger@ logger = ctx.BorrowLogger();
    if (logger !is null) {
      const bool reloadPhaseOk = ctx.IsReloading && ctx.ReloadPhase == BML::RELOAD_LOAD;
      logger.Info("BML state reload phase: v2 load=" + (reloadPhaseOk ? "reload" : "unexpected"));
      logger.Info("BML state reload smoke v2 loaded migrated=" + (migrated ? "true" : "false") +
                  " from=" + fromVersion +
                  " counter=" + counter +
                  " text=" + text);
    }
    InstallServices(ctx);
  }

  void OnProcess(const BML::ModContext &in ctx) {
    ++frames;
    if (!requestedCommand) {
      requestedCommand = true;
      ctx.ExecuteCommand("bml_state_reload_probe");
    }
    if (!publishedDataShare) {
      publishedDataShare = true;
      const bool published = BML::DataShareSetString(
          "bml.state.reload.pending", "from-v2", "BML");
      ctx.LogInfo("BML state reload datashare publish: v2=" +
                  (published ? "valid" : "invalid"));
    }
    ImGui::SetNextWindowPos(ImVec2(64.0f, 64.0f), ImGuiCond_Once);
    ImGui::SetNextWindowSize(ImVec2(320.0f, 0.0f), ImGuiCond_Once);
    if (ImGui::Begin("State Reload Smoke")) {
      ImGui::TextUnformatted("runtime v2");
      ImGui::TextUnformatted("migrated=" + (migrated ? "true" : "false"));
      ImGui::TextUnformatted("counter=" + counter);
    }
    ImGui::End();
    if (frames >= 30 && !requestedExit) {
      requestedExit = true;
      ctx.ExecuteCommand("exit");
    }
  }

  void OnUnload(const BML::ModContext &in ctx) {
    BML::Logger@ logger = ctx.BorrowLogger();
    if (logger !is null) {
      const bool shutdownPhaseOk = !ctx.IsReloading && ctx.ReloadPhase == BML::RELOAD_NONE;
      logger.Info("BML state reload phase: v2 unload=" + (shutdownPhaseOk ? "shutdown" : "unexpected"));
    }
  }

  private void InstallServices(const BML::ModContext &in ctx) {
    BML::TimerLoopCallback@ timerCallback = BML::TimerLoopCallback(this.OnReloadTimer);
    @reloadTimer = ctx.SetIntervalTicks(1, timerCallback, "state-reload-v2");

    BML::CommandDefinition commandDefinition;
    commandDefinition.Name = "bml_state_reload_probe";
    commandDefinition.Description = "Exercise script service ownership during hot reload";
    commandDefinition.Enabled = true;
    BML::CommandCallback@ commandCallback = BML::CommandCallback(this.OnReloadCommand);
    @reloadCommand = ctx.RegisterCommand(commandDefinition, commandCallback);

    BML::DataShareCallback@ dataShareCallback = BML::DataShareCallback(this.OnReloadDataShare);
    @reloadDataShare = ctx.RequestDataShare(
        "bml.state.reload.pending", BML::DATASHARE_STRING, dataShareCallback, "BML");

    const bool timerValid = reloadTimer !is null && reloadTimer.IsValid;
    const bool commandValid = reloadCommand !is null && reloadCommand.IsValid;
    const bool dataShareValid = reloadDataShare !is null && reloadDataShare.IsValid;
    ctx.LogInfo("BML state reload services: v2 timer=" + (timerValid ? "valid" : "invalid") +
                " command=" + (commandValid ? "valid" : "invalid") +
                " datashare=" + (dataShareValid ? "valid" : "invalid"));
  }

  private bool OnReloadTimer(const BML::ModContext &in ctx,
                             const BML::TimerEvent &in event) {
    ctx.LogInfo("BML state reload timer callback: v2");
    return true;
  }

  private void OnReloadCommand(const BML::ModContext &in ctx,
                               const BML::CommandEvent &in event) {
    ctx.LogInfo("BML state reload command callback: v2");
  }

  private void OnReloadDataShare(const BML::ModContext &in ctx,
                                 const BML::DataShareEvent &in event) {
    const bool valid = event.Exists && event.StringValue == "from-v2";
    ctx.LogInfo("BML state reload datashare callback: v2=" +
                (valid ? "valid" : "invalid"));
  }
}
